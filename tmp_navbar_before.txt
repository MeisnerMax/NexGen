import { useState, useEffect, useRef } from 'react';
import { useRouter } from 'next/router';
import Head from 'next/head';
import Image from 'next/image';
import logo from '../public/images/logo.png';

export default function NavBar() {
  return (
    <>
      <Head>
        {/* Open Graph Meta Tags */}
        <meta property="og:type" content="website" />
        <meta property="og:title" content="Nexgen Consulting - Digitalisierung, Webdesign und Automatisierung" />
        <meta property="og:description" content="Nexgen Consulting - Ihr Partner für Digitalisierung, Prozessautomatisierung, Webdesign und SEO-Optimierung für kleine und mittelständische Unternehmen." />
        <meta property="og:image" content="https://nexgen-consulting.de/logo.png" />
        <meta property="og:url" content="https://nexgen-consulting.de" />
        <meta property="og:site_name" content="Nexgen Consulting" />
      </Head>
      <nav className="sticky top-0 left-0 w-full z-50 bg-brand-primary/90 backdrop-blur supports-[backdrop-filter]:bg-brand-primary/80 text-white border-b border-white/10">
        <div className="container mx-auto flex items-center p-4">
          {/* Linke Seite: Brand */}
          <div className="flex items-center space-x-4">
            <Image
              src={logo}
              alt="Nexgen Consulting Logo"
              width={50}
              height={50}
              className="mr-2"
            />
            <Link href="/" className="text-2xl text-brand-accent font-bold hover:underline">
              Nexgen
            </Link>
            <Link href="/" className="text-2xl font-bold hover:underline">
              Consulting
            </Link>
          </div>

          {/* Drawer-Auslöser für alle Bildschirmgrößen */}
          <div className="ml-auto">
            <MobileMenu />
          </div>
        </div>
      </nav>
    </>
  );
}

function MobileMenu() {
  const router = useRouter();
  const [open, setOpen] = useState(false);
  const panelRef = useRef(null);

  const closeMenu = () => setOpen(false);

  const isActive = (href) => {
    const p = router.asPath || '/';
    return p === href || p.startsWith(href + '/');
  };

  const linkClasses = (href) =>
    `group flex items-center justify-between rounded-xl px-3 py-3 text-base transition ${
      isActive(href) ? 'bg-white/10 ring-1 ring-white/10' : 'hover:bg-white/5'
    }`;

  const iconClasses = (href) =>
    `h-5 w-5 ${isActive(href) ? 'text-brand-accent' : 'text-white/60 group-hover:text-brand-accent'}`;

  const Icon = ({ name, className = '' }) => {
    const common = `flex-none ${className}`;
    switch (name) {
      case 'menu':
        return (
          <svg viewBox="0 0 24 24" fill="currentColor" aria-hidden="true" className={common}>
            <path d="M4 6h16v2H4V6zm0 5h16v2H4v-2zm0 5h16v2H4v-2z" />
          </svg>
        );
      case 'home':
        return (
          <svg viewBox="0 0 24 24" fill="currentColor" aria-hidden="true" className={common}>
            <path d="M3 10.5 12 3l9 7.5V21a1.5 1.5 0 0 1-1.5 1.5H15v-6H9v6H4.5A1.5 1.5 0 0 1 3 21v-10.5z" />
          </svg>
        );
      case 'bulb':
        return (
          <svg viewBox="0 0 24 24" fill="currentColor" aria-hidden="true" className={common}>
            <path d="M12 2a7 7 0 0 0-4 12.9V17a2 2 0 0 0 2 2h4a2 2 0 0 0 2-2v-2.1A7 7 0 0 0 12 2zm-2 18a2 2 0 0 0 4 0h-4z" />
          </svg>
        );
      case 'globe':
        return (
          <svg viewBox="0 0 24 24" fill="currentColor" aria-hidden="true" className={common}>
            <path d="M12 2a10 10 0 1 0 0 20 10 10 0 0 0 0-20zm6.9 6h-3.2A15.8 15.8 0 0 0 13 4.3 8.04 8.04 0 0 1 18.9 8zM12 4.1c.9.7 2 2.2 2.6 3.9H9.4c.6-1.7 1.7-3.2 2.6-3.9zM7 12c0-.7.1-1.4.3-2h9.4c.2.6.3 1.3.3 2s-.1 1.4-.3 2H7.3A8 8 0 0 1 7 12zm.1 4h3.2A15.8 15.8 0 0 0 11 19.7 8.04 8.04 0 0 1 7.1 16zM12 19.9c-.9-.7-2-2.2-2.6-3.9h5.2c-.6 1.7-1.7 3.2-2.6 3.9zM12 4.3A15.8 15.8 0 0 0 9.9 8H6.1A8.04 8.04 0 0 1 12 4.3zM5.1 16A8.04 8.04 0 0 1 3.9 12c0-.7.1-1.4.3-2h3.8c-.2.6-.3 1.3-.3 2s.1 1.4.3 2H5.1zM14.1 16h3.8a8.04 8.04 0 0 1-1.2 4 15.8 15.8 0 0 0-2.6-4z" />
          </svg>
        );
      case 'megaphone':
        return (
          <svg viewBox="0 0 24 24" fill="currentColor" aria-hidden="true" className={common}>
            <path d="M3 11a2 2 0 0 0 2 2h2l8 4V5l-8 4H5a2 2 0 0 0-2 2zm8 6v2a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2v-2h6z" />
          </svg>
        );
      case 'id':
        return (
          <svg viewBox="0 0 24 24" fill="currentColor" aria-hidden="true" className={common}>
            <path d="M4 4h16a2 2 0 0 1 2 2v12a2 2 0 0 1-2 2H4a2 2 0 0 1-2-2V6a2 2 0 0 1 2-2zm4 4a3 3 0 1 0 0 6 3 3 0 0 0 0-6zm6 1h6v2h-6V9zm0 4h6v2h-6v-2z" />
          </svg>
        );
      case 'puzzle':
        return (
          <svg viewBox="0 0 24 24" fill="currentColor" aria-hidden="true" className={common}>
            <path d="M8 2a2 2 0 0 1 2 2v1h2a2 2 0 1 1 0 4h-2v2h2a2 2 0 1 1 0 4h-2v1a2 2 0 0 1-4 0v-1H5a2 2 0 1 1 0-4h1V9H5a2 2 0 1 1 0-4h1V4a2 2 0 0 1 2-2z" />
          </svg>
        );
      case 'code':
        return (
          <svg viewBox="0 0 24 24" fill="currentColor" aria-hidden="true" className={common}>
            <path d="M9 18 3 12l6-6 1.5 1.5L6 12l4.5 4.5L9 18zm6 0-1.5-1.5L18 12l-4.5-4.5L15 6l6 6-6 6z" />
          </svg>
        );
      case 'phone':
        return (
          <svg viewBox="0 0 24 24" fill="currentColor" aria-hidden="true" className={common}>
            <path d="M7 2h10a2 2 0 0 1 2 2v16a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2V4a2 2 0 0 1 2-2zm3 18h4v-1h-4v1z" />
          </svg>
        );
      case 'newspaper':
        return (
          <svg viewBox="0 0 24 24" fill="currentColor" aria-hidden="true" className={common}>
            <path d="M4 5h14a2 2 0 0 1 2 2v11H6a2 2 0 0 1-2-2V5zm0 13a2 2 0 0 1-2-2V8h2v10zm4-9h10v3H8V9zm0 4h10v2H8v-2z" />
          </svg>
        );
      default:
        return null;
    }
  };

  // ESC zum Schließen (kein Body-Scroll-Lock)
  useEffect(() => {
    const onKey = (e) => {
      if (e.key === 'Escape') closeMenu();
    };
    document.addEventListener('keydown', onKey);
    return () => document.removeEventListener('keydown', onKey);
  }, []);

  // Fokus in den Drawer setzen
  useEffect(() => {
    if (open && panelRef.current) {
      const firstLink = panelRef.current.querySelector('a, button');
      firstLink?.focus?.();
    }
  }, [open]);

  return (
    <div className="relative">
      <button
        className="inline-flex items-center gap-2 rounded-lg border border-white/15 bg-white/5 px-3 py-2 text-sm font-medium text-white hover:bg-white/10 active:bg-white/15 transition focus:outline-none focus:ring-2 focus:ring-brand-accent/60"
        onClick={() => setOpen(true)}
        aria-label="Menü öffnen"
        aria-expanded={open}
        aria-controls="mobile-drawer"
      >
        <span className="sr-only">Menü</span>
        <span aria-hidden>☰</span>
        <span aria-hidden className="hidden sm:inline">Menü</span>
      </button>

      {/* Overlay: dimmt, aber lässt Scroll zu */}
      <div
        className={`fixed inset-0 z-[60] bg-black/40 backdrop-blur-sm transition-opacity duration-300 ease-brand ${open ? 'opacity-100' : 'pointer-events-none opacity-0'}`}
        onClick={closeMenu}
        aria-hidden={!open}
      />

      {/* Drawer Panel mit deckendem Hintergrund und verbessertem Header */}
      <aside
        id="mobile-drawer"
        role="dialog"
        aria-modal="true"
        aria-label="Hauptmenü"
        ref={panelRef}
        className={`fixed inset-y-0 right-0 z-[70] w-[88%] max-w-sm bg-brand-primary shadow-overlay border-l border-white/10 px-5 py-6 flex flex-col transition-transform duration-300 ease-brand ${open ? 'translate-x-0' : 'translate-x-full'}`}
      >
        {/* Header im Drawer */}
        <div className="flex items-center justify-between pb-3 mb-2 border-b border-white/10">
          <div className="flex items-center gap-3">
            <Icon name="menu" className="h-6 w-6 text-brand-accent" />
            <div className="leading-tight">
              <div className="font-heading text-lg">Navigation</div>
              <div className="text-xs text-white/60">Schnellzugriff auf alle Bereiche</div>
            </div>
          </div>
          <button
            onClick={closeMenu}
            className="rounded-lg p-2 text-white/90 hover:text-brand-accent hover:bg-white/10 focus:outline-none focus:ring-2 focus:ring-brand-accent/60"
            aria-label="Menü schließen"
          >
            ✕
          </button>
        </div>

        {/* Links */}
        <nav className="mt-2 grid gap-1.5" onClick={closeMenu}>
          <Link href="/" className={linkClasses('/')}>
            <span className="flex items-center gap-3">
              <Icon name="home" className={iconClasses('/')} />
              <span>Home</span>
            </span>
            <span className="text-brand-accent/60 group-hover:text-brand-accent">→</span>
          </Link>

          <div className="mt-2 mb-1 px-3 text-xs uppercase tracking-wide text-white/50">Leistungen</div>
          <Link href="/services/beratung" className={linkClasses('/services/beratung')}>
            <span className="flex items-center gap-3">
              <Icon name="bulb" className={iconClasses('/services/beratung')} />
              <span>Beratung</span>
            </span>
            <span className="text-brand-accent/60 group-hover:text-brand-accent">→</span>
          </Link>
          <Link href="/services/website" className={linkClasses('/services/website')}>
            <span className="flex items-center gap-3">
              <Icon name="globe" className={iconClasses('/services/website')} />
              <span>Website</span>
            </span>
            <span className="text-brand-accent/60 group-hover:text-brand-accent">→</span>
          </Link>
          <Link href="/services/marketing" className={linkClasses('/services/marketing')}>
            <span className="flex items-center gap-3">
              <Icon name="megaphone" className={iconClasses('/services/marketing')} />
              <span>Marketing</span>
            </span>
            <span className="text-brand-accent/60 group-hover:text-brand-accent">→</span>
          </Link>
          <Link href="/services/firmenidentitaet" className={linkClasses('/services/firmenidentitaet')}>
            <span className="flex items-center gap-3">
              <Icon name="id" className={iconClasses('/services/firmenidentitaet')} />
              <span>Firmenidentität</span>
            </span>
            <span className="text-brand-accent/60 group-hover:text-brand-accent">→</span>
          </Link>
          <Link href="/services/loesungen" className={linkClasses('/services/loesungen')}>
            <span className="flex items-center gap-3">
              <Icon name="puzzle" className={iconClasses('/services/loesungen')} />
              <span>Lösungen</span>
            </span>
            <span className="text-brand-accent/60 group-hover:text-brand-accent">→</span>
          </Link>
          <Link href="/services/softwareentwicklung" className={linkClasses('/services/softwareentwicklung')}>
            <span className="flex items-center gap-3">
              <Icon name="code" className={iconClasses('/services/softwareentwicklung')} />
              <span>Softwareentwicklung</span>
            </span>
            <span className="text-brand-accent/60 group-hover:text-brand-accent">→</span>
          </Link>
          <Link href="/services/app" className={linkClasses('/services/app')}>
            <span className="flex items-center gap-3">
              <Icon name="phone" className={iconClasses('/services/app')} />
              <span>App-Entwicklung</span>
            </span>
            <span className="text-brand-accent/60 group-hover:text-brand-accent">→</span>
          </Link>

          <div className="mt-4 mb-1 px-3 text-xs uppercase tracking-wide text-white/50">Ressourcen</div>
          <Link href="/blog" className={linkClasses('/blog')}>
            <span className="flex items-center gap-3">
              <Icon name="newspaper" className={iconClasses('/blog')} />
              <span>Blog</span>
            </span>
            <span className="text-brand-accent/60 group-hover:text-brand-accent">→</span>
          </Link>
        </nav>

        {/* Footer-CTA & Socials */}
        <div className="mt-auto pt-6 border-t border-white/10">
          <Link
            href="/services/contact"
            className="inline-flex w-full items-center justify-center gap-2 rounded-xl bg-brand-accent px-4 py-3 font-medium text-brand-primary shadow-card hover:opacity-95 active:opacity-90 transition"
          >
            Kontakt aufnehmen
          </Link>
          <div className="mt-4 flex items-center justify-center gap-6 text-sm text-white/80">
            <a href="https://www.linkedin.com/company/106936390" target="_blank" rel="noopener noreferrer" className="hover:text-white">LinkedIn</a>
            <a href="https://www.instagram.com/nexgenconsultingcoburg/" target="_blank" rel="noopener noreferrer" className="hover:text-white">Instagram</a>
          </div>
        </div>
      </aside>
    </div>
  );
}
